  author_name VARCHAR(100),
  author_title VARCHAR(100),
  author_company VARCHAR(100),
  industry VARCHAR(50),
  metric_type VARCHAR(50),
  metric_value VARCHAR(50),
  category VARCHAR(50),
  length VARCHAR(20),
  tone VARCHAR(20),
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
